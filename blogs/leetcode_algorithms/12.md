- Sorting using constraints of the values between certain min and max values

## Code (Python)
```python
class Solution:
   def maximumGap(self, nums: List[int]) -> int:
       mmax = max(nums)
       mmin = min(nums)
       n = len(nums)

       if n < 2:
           return 0

       bucketSize = max((mmax - mmin) // (n - 1), 1)
       buckets = defaultdict(list)
      
       for num in nums:
           key = (num - mmin) // bucketSize
           if not buckets[key]:
               buckets[key] = [num, num]
           else:
               buckets[key] = [min(buckets[key][0], num), max(buckets[key][1], num)]

       res = 0
       prev = -1

       for key in sorted(buckets.keys()):
           if prev != -1:
               res = max(res, buckets[key][0] - buckets[prev][1])
           prev = key

       return res
```
