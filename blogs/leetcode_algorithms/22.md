- Uses LPS array to match prefix and suffix array (think like shifting needle to find its match in the haystack array)

## Code (Python)
```python
class Solution:
   def strStr(self, haystack: str, needle: str) -> int:
       if needle == "": return 0
       lps = [0] * len(needle)

       prevLPS, i = 0, 1

       # O(2*n)
       while i < len(needle):
           if needle[i] == needle[prevLPS]:
               lps[i] = prevLPS + 1
               prevLPS += 1
               i += 1
           elif prevLPS == 0:
               lps[i] = 0
               i += 1
           else:
               prevLPS = lps[prevLPS - 1]
      
       i, j = 0, 0
       while i < len(haystack):
           if haystack[i] == needle[j]:
               i, j = i + 1, j + 1
           else:
               if j == 0:
                   i += 1
               else:
                   j = lps[j-1]
           if j == len(needle):
               return i - len(needle)
       return -1
```
