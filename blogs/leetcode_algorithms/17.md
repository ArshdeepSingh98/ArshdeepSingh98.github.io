- A graph where its vertices can be divided into two disjoint sets so that all edges connect vertices in different sets

## Code (Python)
- Using BFS
```python
class Solution:
    def possibleBipartition(self, n: int, dislikes: List[List[int]]) -> bool:
        adj = defaultdict(set)


        for n1, n2 in dislikes:
            adj[n1].add(n2)
            adj[n2].add(n1)
       
        color = defaultdict(lambda: -1)


        for node in range(1, n + 1):
            if color[node] == -1:
                q = deque([node])
                color[node] = 0


                while q:
                    node = q.popleft()
                    for nei in adj[node]:
                        if color[nei] == color[node]:
                            return False
                        if color[nei] == -1:
                            color[nei] = 1 - color[node]
                            q.append(nei)
       
        return True
```

- Using Union-Find
```python
class Solution:
    def possibleBipartition(self, n: int, dislikes: List[List[int]]) -> bool:
        par = [i for i in range(n + 2)]
        rank  = [1] * (n + 2)
        adj = defaultdict(list)
       
        for n1, n2 in dislikes:
            adj[n1].append(n2)
            adj[n2].append(n1)
       
        def find(x):
            res = x
            while res != par[res]:
                par[res] = par[par[res]]
                res = par[res]
            return res


        def union(x, y):
            p1, p2 = find(x), find(y)
            if p1 == p2:
                return 0
            elif rank[p1] > rank[p2]:
                rank[p1] += rank[p2]
                par[p2] = p1
            else:
                rank[p2] += rank[p1]
                par[p1] = p2
            return 1
       
        for node in range(1, n + 1):
            for nei in adj[node]:
                if find(node) == find(nei):
                    return False
                union(adj[node][0], nei)
        return True
```
