Helps to detect cycles in graphs or get ordering of nodes in directed graph based on indegree

## Code (Python)
```python
class Solution:
    def buildMatrix(self, k: int, rowConditions: List[List[int]], colConditions: List[List[int]]) -> List[List[int]]:
        def topsort(conditions):
            adj = defaultdict(list)
            indeg = [0] * k


            for t, b in conditions:
                t -= 1
                b -= 1
                adj[t].append(b)
                indeg[t] += 0
                indeg[b] += 1
           
            q = deque()
            for i in range(k):
                if indeg[i] == 0:
                    q.append(i)
           
            res = []
            while q:
                node = q.popleft()
                res.append(node)
                for nei in adj[node]:
                    indeg[nei] -= 1
                    if indeg[nei] == 0:
                        q.append(nei)


            assert(len(res) == k)
            return {x: i for i, x in enumerate(res)}


        try:
            rt = topsort(rowConditions)
            ct = topsort(colConditions)
        except:
            return []
       
        res = [[0] * k for _ in range(k)]
        for num in range(k):
            res[rt[num]][ct[num]] = num + 1
        return res
```
