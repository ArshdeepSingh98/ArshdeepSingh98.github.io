Helps in effective storage and retrieval of words and in prefix based problems

## Code (Python)
```python
class Node:
   def __init__(self):
       self.cl = {}
       self.end = False

class Trie:
   def __init__(self):
       self.root = Node()

   def insert(self, word: str) -> None:
       curr = self.root
       for c in word:
           if c not in curr.cl:
               curr.cl[c] = Node()
           curr = curr.cl[c]
       curr.end = True
      

   def search(self, word: str) -> bool:
       curr = self.root
       for c in word:
           if c not in curr.cl:
               return False
           curr = curr.cl[c]
       return curr.end
      

   def startsWith(self, prefix: str) -> bool:
       curr = self.root
       for c in prefix:
           if c not in curr.cl:
               return False
           curr = curr.cl[c]
       return True
```
