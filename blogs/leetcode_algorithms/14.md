## Code (Python)
```python
N = 9
class Solution:
    def solveSudoku(self, board: List[List[str]]) -> None:
        """
        Do not return anything, modify board in-place instead.
        """
        def valid():
            for r in range(N):
                counts = [0] * 10
                for c in range(N):
                    if board[r][c] == '.':
                        continue
                    num = int(board[r][c])
                    counts[num] += 1
                    if counts[num] > 1:
                        return False
           
            for c in range(N):
                counts = [0] * 10
                for r in range(N):
                    if board[r][c] == '.':
                        continue
                    num = int(board[r][c])
                    counts[num] += 1
                    if counts[num] > 1:
                        return False
           
            # top left
            for r in range(0, 7, 3):
                for c in range(0, 7, 3):
                    counts = [0] * 10
                    for small_r in range(r, r + 3):
                        for small_c in range(c, c + 3):
                            if board[small_r][small_c] == '.':
                                continue
                            num = int(board[small_r][small_c])
                            counts[num] += 1
                            if counts[num] > 1:
                                return False


            return True


        def potential_cands(x, y):
            res = set([str(x) for x in range(1, 10)])
            for c in range(N):
                if board[x][c] != '.':
                    res.discard(board[x][c])
           
            for r in range(N):
                if board[r][y] != '.':
                    res.discard(board[r][y])
           
            for tr in range(x // 3 * 3, x // 3 * 3 + 3):
                for tc in range(y // 3 * 3, y // 3 * 3 + 3):
                    if board[tr][tc] != '.':
                        res.discard(board[tr][tc])
           
            return list(res)
       
        pre_processed_cands = {}
        for r in range(N):
            for c in range(N):
                if board[r][c] == '.':
                    pre_processed_cands[(r,c)] = potential_cands(r, c)


        def util(posn):
            r, c = posn // N, posn % N
            if not valid(): return False
            if posn >= N * N: return True
            if board[r][c] != ".":
                if util(posn + 1): return True
            else:
                for candidate in pre_processed_cands[(r, c)]:
                    board[r][c] = candidate
                    if util(posn + 1): return True
                board[r][c] = "."
            return False
        util(0)
```
