## Code (Python)
- Without Regex
```python
class Solution:
    def countAndSay(self, n: int) -> str:
        prev = "1"


        def RLE(s):
            res = ""
            cnt = 1
            prev = None
            s += "$"
            for c in s:
                if prev is None:
                    prev = c
                else:
                    if c == prev:
                        cnt += 1
                    else:
                        res += f"{cnt}{prev}"
                        cnt = 1
                        prev = c
            return res


        for i in range(2, n+1):
            prev = RLE(prev)
        return prev
```
