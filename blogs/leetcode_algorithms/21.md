class SegmentTree:
    def __init__(self, n):
        self.n = n
        self.tree = [0] * (4 * n)
        self.lazy = [None] * (4 * n)
        self.pending_update = [False] * (4 * n)


    def _build(self, arr, start, end, node):
        if start == end:
            self.tree[node] = arr[start]
        else:
            mid = (start + end) // 2
            self._build(arr, start, mid, 2 * node + 1)
            self._build(arr, mid + 1, end, 2 * node + 2)
            self.tree[node] = self.tree[2 * node + 1] + self.tree[2 * node + 2]


    def build(self, arr):
        self._build(arr, 0, self.n - 1, 0)


    def _update_range(self, start, end, l, r, value, node):
        if self.pending_update[node]:
            self.tree[node] = (end - start + 1) * self.lazy[node]
            if start != end:
                self.lazy[2 * node + 1] = self.lazy[node]
                self.lazy[2 * node + 2] = self.lazy[node]
                self.pending_update[2 * node + 1] = True
                self.pending_update[2 * node + 2] = True
            self.pending_update[node] = False


        if start > end or start > r or end < l:
            return


        if start >= l and end <= r:
            self.tree[node] = (end - start + 1) * value
            if start != end:
                self.lazy[2 * node + 1] = value
                self.lazy[2 * node + 2] = value
                self.pending_update[2 * node + 1] = True
                self.pending_update[2 * node + 2] = True
            return


        mid = (start + end) // 2
        self._update_range(start, mid, l, r, value, 2 * node + 1)
        self._update_range(mid + 1, end, l, r, value, 2 * node + 2)
        self.tree[node] = self.tree[2 * node + 1] + self.tree[2 * node + 2]


    def update_range(self, l, r, value):
        self._update_range(0, self.n - 1, l, r, value, 0)


    def _query_range(self, start, end, l, r, node):
        if self.pending_update[node]:
            self.tree[node] = (end - start + 1) * self.lazy[node]
            if start != end:
                self.lazy[2 * node + 1] = self.lazy[node]
                self.lazy[2 * node + 2] = self.lazy[node]
                self.pending_update[2 * node + 1] = True
                self.pending_update[2 * node + 2] = True
            self.pending_update[node] = False


        if start > end or start > r or end < l:
            return 0


        if start >= l and end <= r:
            return self.tree[node]


        mid = (start + end) // 2
        left_sum = self._query_range(start, mid, l, r, 2 * node + 1)
        right_sum = self._query_range(mid + 1, end, l, r, 2 * node + 2)
        return left_sum + right_sum


    def query_range(self, l, r):
        return self._query_range(0, self.n - 1, l, r, 0)


class Solution:
    def amountPainted(self, paint: List[List[int]]) -> List[int]:
        N = len(paint)
        d = [1] * 10**5
        # when painted, make it 0
        # query search for number of 1's in the interval, exluding the end


        seg_tree = SegmentTree(len(d))
        seg_tree.build(d)
        res = []


        for s, e in paint:
            sum_in_range = seg_tree.query_range(s, e-1)
            res.append(sum_in_range)
            seg_tree.update_range(s, e-1, 0)


        return res
