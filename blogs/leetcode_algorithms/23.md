- Convert target to hash
- Convert each window to hash, and slide the window, update the hash values and match
- Effective matching in sliding window for strings, helps find the target string in source string faster

## Code (Python)
```python
class Solution:
    def rotateString(self, s: str, goal: str) -> bool:
        if len(s) != len(goal): return False
        s = s + s
        base = 29
        p = 1
        curr = 0
        mod = 10**9


        for c in goal:
            curr = (curr * base + ord(c) - ord('a'))%mod
            p = p * base
       
        target = curr
        curr = 0
        l = 0
        for i, c in enumerate(s):
            curr = (curr * base + ord(c) - ord('a'))%mod
            l += 1
            if l > len(goal):
                curr = (curr - p * (ord(s[i-len(goal)]) - ord('a')))%mod
            if curr == target:
                return True


        return False

class Solution:
    def findPattern(self, stream: Optional['InfiniteStream'], pattern: List[int]) -> int:
        curr = 0
        base_pow = 1
        base = 2
        mod = 10**9 + 7
        P = len(pattern)


        for i, c in enumerate(pattern):
            curr = (curr * base + c) % mod
            base_pow = (base_pow * base) % mod
       
        index = 0
        window = []
        win_hash = 0
        while True:
            val = stream.next()
            window.append(val)
            win_hash = (win_hash * base + val) % mod


            if len(window) > P:
                win_hash = (win_hash - window[0] * base_pow) % mod
                del window[0]
            if win_hash == curr and index >= P-1:
                return index - P + 1
            index += 1


        assert(False)
```
