## Code (Python)
```python
class Solution:
    def reverseParentheses(self, s: str) -> str:
        res = ""
        pairs = {}
        stack = []
        for i, c in enumerate(s):
            if c == '(':
                stack.append(i)
            elif c == ')':
                pairs[stack[-1]] = i
                pairs[i] = stack[-1]
                stack.pop()


        i, dir = 0, 1
        n = len(s)
        while i < n:
            if s[i] == '(' or s[i] == ')':
                i = pairs[i]
                dir *= -1
            else:
                res += s[i]
            i = i + dir


        return res  
```
