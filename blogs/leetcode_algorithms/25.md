```python
class Solution:
    def sortColors(self, nums: List[int]) -> None:
        """
        Do not return anything, modify nums in-place instead.
        """
        l = i = 0
        r = len(nums) - 1

        # [2(l),0,2,1,1,2(r)]
        # [2(l),0,2,1,1(r),2]
        # [1(l),0,2,1(r),2,2]
        # [1,0(l),2,1(r),2,2]
        # [0,1,2(l),1(r),2,2]
        # [0,1,1(l),2(r),2,2]

        while i <= r:
            if nums[i] == 0:
                nums[l], nums[i] = nums[i], nums[l]
                l += 1
                i += 1
            elif nums[i] == 2:
                nums[i], nums[r] = nums[r], nums[i]
                r -= 1
            else:
                i += 1
```
