Find smallest distances between each pair of nodes in a graph

## Code (Python)
```python
class Solution:
    def findTheCity(self, n: int, edges: List[List[int]], threshold: int) -> int:
        d = [[inf] * n for _ in range(n)]


        for i in range(n):
            d[i][i] = 0
       
        for u, v, w in edges:
            d[u][v] = w
            d[v][u] = w


        for k in range(n):
            for i in range(n):
                for j in range(n):
                    d[i][j] = min(d[i][j], d[i][k] + d[k][j])
               
        res = None
        min_count = n + 1
        for i in range(n):
            count = 0
            for j in range(n):
                if d[i][j] <= threshold:
                    count += 1
            if count <= min_count:
                min_count = count
                res = i


        return res
```
