- O(1) Space complexity inorder traversal of binary tree


## Code (Python)
```python
class Solution:
    def inorderTraversal(self, root: Optional[TreeNode]) -> List[int]:
        res = []
        curr = root


        # morris traversal
        while curr:
            if not curr.left:
                res.append(curr.val)
                curr = curr.right # this will use thread for inorder
            else:
                left_node = curr.left
                while left_node.right and left_node.right != curr:
                    left_node = left_node.right
               
                # if thread is not formed yet
                if not left_node.right:
                    left_node.right = curr
                    curr = curr.left
                else:
                    # thread is formed, delete it and add current node to inorder
                    left_node.right = None
                    res.append(curr.val)
                    curr = curr.right
       
        return res

```
