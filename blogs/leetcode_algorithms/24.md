- Effective calc of range values, using Prefix Sum

## Code (Python)
```python
class Solution:
    def maxSumRangeQuery(self, nums: List[int], requests: List[List[int]]) -> int:
        nums.sort()
        n = len(nums)
        mod = 10**9 + 7
        freq = defaultdict(int)
        for l, r in requests:
            freq[l] += 1
            freq[r + 1] -= 1
        for i in range(1, n + 1):
            freq[i] += freq[i-1]
        w = sorted([freq[i] for i in range(n)])
        res = 0
        for i in range(n):
            res += nums[i] * w[i]


        return res % mod

class Solution:
    def numberOfPoints(self, nums: List[List[int]]) -> int:
        ar = [0] * 102


        for l, r in nums:
            ar[l] += 1
            ar[r+1] -= 1
       
        res = 0
        for i in range(1, 101):
            ar[i] += ar[i-1]
            if ar[i] > 0:
                res += 1
        return res
```
