## Code (Python)
```python
class Solution:
   def maximumGap(self, nums: List[int]) -> int:
       if len(nums) < 2:
           return 0
      
       mmax = max(nums)
       exp = 1
       base = 10

       while mmax // exp > 0:
           count = [0] * base
           for n in nums:
               count[ (n // exp) % 10] += 1
           for i in range(1, base):
               count[i] += count[i-1]
          
           aux = [0] * len(nums)
           for i in range(len(nums) - 1, -1, -1):
               aux[count[(nums[i] // exp) % 10] - 1] = nums[i]
               count[(nums[i] // exp) % 10] -= 1

           nums = aux.copy()
           exp *= 10
      
       res = 0
       prev = None
       for num in nums:
           if prev:
               res = max(res, num - prev)
           prev = num

       return res
```
