## Code (Python)
```python
class Solution:
    def longestSubarray(self, nums: List[int], limit: int) -> int:
        max_d = deque()
        min_d = deque()
        l = 0
        res = 0


        for r in range(len(nums)):
            while max_d and max_d[-1] < nums[r]:
                max_d.pop() # small numbers are popped
            max_d.append(nums[r]) # dec order (largest in front)
       
            while min_d and min_d[-1] > nums[r]:
                min_d.pop() # larger numbers are popped
            min_d.append(nums[r]) # inc order (smallest in front)


            while max_d[0] - min_d[0] > limit:
                if max_d[0] == nums[l]:
                    max_d.popleft()
                if min_d[0] == nums[l]:
                    min_d.popleft()
                l += 1
            res = max(res, r - l + 1)
        return res
```
