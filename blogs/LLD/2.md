class Twitter:


    def __init__(self):
        # fol: map of userId to following users
        # twe: map of userId to tweets sorted by recents
        self.fol = defaultdict(set)
        self.twe = defaultdict(deque)
        self.time = 0
       


    def postTweet(self, userId: int, tweetId: int) -> None:
        if len(self.twe[userId]) > 10:
            self.twe[userId].popleft()
        self.twe[userId].append((tweetId, self.time))
        self.time += 1


    def _merge(self, ids):
        heap = []
        for user_id in ids:
            for tweet_id, time in self.twe[user_id]:
                heappush(heap, (time, tweet_id))
                if len(heap) > 10:
                    top = heappop(heap)
        res = []
        while heap:
            time, tweet_id = heappop(heap)
            res.append(tweet_id)
        return res[::-1]


    def getNewsFeed(self, userId: int) -> List[int]:
        allIds = [userId]
        allIds.extend(list(self.fol[userId]))
        res = self._merge(allIds)
        return res
       


    def follow(self, followerId: int, followeeId: int) -> None:
        self.fol[followerId].add(followeeId)
       


    def unfollow(self, followerId: int, followeeId: int) -> None:
        self.fol[followerId].discard(followeeId)
