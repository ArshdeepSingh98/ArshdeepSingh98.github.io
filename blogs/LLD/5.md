## Code (Python)
```python
class Node:
    def __init__(self, val, idx=None, prev=None, next=None):
        self.val = val
        self.next = next


class MyCircularQueue:


    def __init__(self, k: int):
        self.front = None
        self.end = None
        self.N = k
        self.csize = 0


    def enQueue(self, value: int) -> bool:
        # check if full
        if self.isFull(): return False
       
        # update end
        if not self.front:
            # if no front
            self.front = Node(value)
            self.end = self.front
            self.csize = 1
        else:
            # front is present
            self.end.next = Node(value)
            self.end = self.end.next
            self.csize += 1


        return True


    def deQueue(self) -> bool:
        if self.isEmpty(): return False
       
        # update front
        if self.front == self.end:
            self.end = None
            self.front = None
            self.csize = 0
            return True
           
        tmp = self.front.next
        del self.front
        self.front = tmp
        self.csize -= 1
       
        return True


    def Front(self) -> int:
        return self.front.val if self.front else -1


    def Rear(self) -> int:
        return self.end.val if self.end else -1


    def isEmpty(self) -> bool:
        return self.csize == 0


    def isFull(self) -> bool:
        return self.csize == self.N
```
