class SnakeGame:


    def __init__(self, width: int, height: int, food: List[List[int]]):
        self.grid = [[0] * width for _ in range(height)]
        self.food = food
        self.snake = set([(0, 0)])
        self.snakeq = deque([[0, 0]])
        self.head = [0, 0]
        self.score = 0


    def move(self, direction: str) -> int:
        if direction == "R":
            self.head[1] += 1
        elif direction == "U":
            self.head[0] -= 1
        elif direction == "D":
            self.head[0] += 1
        else:
            self.head[1] -= 1
       
        if self.food and self.head == self.food[0]:
            self.score += 1
            self.food.pop(0)
        self.snakeq.append(self.head.copy())
       
        if len(self.snakeq) > self.score + 1:
            tmp = self.snakeq.popleft()
            self.snake.remove((tmp[0], tmp[1]))
       
        if (self.head[0], self.head[1]) in self.snake:
            return -1
        self.snake.add((self.head[0], self.head[1]))
       
        if (self.head[0] < 0 or
            self.head[0] >= len(self.grid) or
            self.head[1] < 0 or
            self.head[1] >= len(self.grid[0])):
            return -1
        return self.score
