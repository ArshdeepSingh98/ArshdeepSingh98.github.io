BASE62_ALPHABET = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ"


class Codec:
    def __init__(self):
        self.string_to_id = {}
        self.id_to_string = {}
        self.counter = 1


    def base62_encode(self, num):
        """Encodes a number in Base62."""
        if num == 0:
            return BASE62_ALPHABET[0]


        encoded = []
        base = len(BASE62_ALPHABET)
        while num:
            num, rem = divmod(num, base)
            encoded.append(BASE62_ALPHABET[rem])
        encoded.reverse()


        return ''.join(encoded)


    def base62_decode(self, base62_string):
        """Decodes a Base62 string into a number."""
        decoded = 0
        base = len(BASE62_ALPHABET)
        for char in base62_string:
            decoded = decoded * base + BASE62_ALPHABET.index(char)
        return decoded


    def encode(self, longUrl: str) -> str:
