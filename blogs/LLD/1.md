class HitCounter:


    def __init__(self):
        self.hits = []


    def hit(self, timestamp: int) -> None:
        self.hits.append(timestamp)


    def getHits(self, timestamp: int) -> int:
        r = bisect_right(self.hits, timestamp - 300)
        return len(self.hits) - r
